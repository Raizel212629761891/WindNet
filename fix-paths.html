Wind Net PC Builder - Image Path Fixer

This tool will fix image path issues in the PC Builder application. Click the button below to apply the fix to your current page.

Results will appear here after running the fix.

Copy this code to your browser console

// Fix image paths at runtime
function fixImagePaths() {
    console.log('Running image path fix...');
    // Find all image elements
    const images = document.querySelectorAll('img');
    let fixedCount = 0;
    
    // Process each image
    images.forEach(img => {
        let src = img.getAttribute('src');
        if (src) {
            let originalSrc = src;
            
            // Fix double category in path
            const categories = ['monitor', 'hard drive', 'cpu cooler', 'graphics card'];
            categories.forEach(category => {
                // Create pattern for capitalized category name
                const capitalizedCategory = category.split(' ')
                    .map(word => word.charAt(0).toUpperCase() + word.slice(1))
                    .join(' ');
                
                // Check for duplicate category in path
                if (src.includes(`components/${category}/${capitalizedCategory}/`)) {
                    src = src.replace(`components/${category}/${capitalizedCategory}/`, `components/${category}/`);
                }
            });
            
            // Fix backslashes
            src = src.replace(/\\/g, '/');
            
            // Update the image source if it changed
            if (src !== originalSrc) {
                console.log(`Updated image path from ${originalSrc} to ${src}`);
                img.setAttribute('src', src);
                fixedCount++;
            }
        }
    });
    
    return fixedCount;
}

// Run the fix
const fixedCount = fixImagePaths();
console.log(`Fixed ${fixedCount} image paths`);

// Add event listener to fix paths for dynamically loaded images
document.addEventListener('DOMContentLoaded', function() {
    // Fix on page load
    fixImagePaths();
    
    // Fix when new content is added
    const observer = new MutationObserver(function() {
        fixImagePaths();
    });
    
    // Start observing
    observer.observe(document.body, { 
        childList: true, 
        subtree: true 
    });
});
